# 使用多继承与虚函数实现返回基类共享指针与运行时多态

```CPP
class NodeBaseInterface
{
public:
    NodeBaseInterface() = default;
    ~NodeBaseInterface() = default;
    virtual void create_callback_group() const {
        std::cout << "call NodeBaseInterface::create_callback_group\r\n";
        return;
    }
};

class NodeBase : public NodeBaseInterface,public std::enable_shared_from_this<NodeBase>
{
public:
    NodeBase() = default;
    ~NodeBase() = default;
    void create_callback_group() const override {
        std::cout << "call NodeBase::create_callback_group\r\n";
        return;
    }
    std::shared_ptr<NodeBaseInterface> get_node_base_interface() {
        return this->shared_from_this();
    }
};
```
